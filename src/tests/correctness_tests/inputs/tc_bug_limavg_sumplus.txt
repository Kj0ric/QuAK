# Bug exposure test: LimSupAvg + SumPlus with mixed-sign weights.
#
# If the projection shortcut fires for LimSupAvg (it shouldn't), it calls
# isNonEmpty(LimSupAvg, SumB, x=5, bound=5), which uses flatten_regular(SumB, 5).
# SumB(5) caps child-1's projected value (8) to 5, giving LimSupAvg([5,4,...]) = 4.5 < 5 → FALSE.
#
# The correct algorithm should project first, then run the LimSupAvg+SumPlus
# fast/slow path with theoretical_bound (not x), giving LimSupAvg([8,4,...]) = 6 >= 5 → TRUE.
#
# Parent: deterministic, alternates child 1 and child 2 on unary alphabet {a}.
# Child 1: weights [+6, -2], SumPlus = |6| + |-2| = 8  (mixed-sign -- triggers projection)
# Child 2: weights [+3, +1], SumPlus = |3| + |1|  = 4  (positive-only)
#
# Sequence of child outputs: [8, 4, 8, 4, ...], LimSupAvg = 6
#
# Expected:
#   non-empty LimSupAvg SumPlus 5  -> TRUE  (LimSupAvg = 6 >= 5)
#   non-empty LimSupAvg SumPlus 7  -> FALSE (LimSupAvg = 6 < 7)

@PARENT
a : 1, p0 -> p1
a : 2, p1 -> p0

@CHILD 0

@CHILD 1
final: f
a :  6, c0 -> c1
a : -2, c1 -> f

@CHILD 2
final: f
a :  3, c0 -> c1
a :  1, c1 -> f
