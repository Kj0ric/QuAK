#
# On a^omega the first a starts Child 1, which can terminate only on b.
# Every b-step leaves the accepting parent region forever. The later a-steps in
# q1 can start Child 2 with SumPlus value 1, but the original Child 1
# background obligation never terminates along any accepted word.
#
# Expected at threshold 1:
# - Sup x SumPlus: false
# - LimSup x SumPlus: false

final: q1
a : 1, q0 -> q1
b : 0, q0 -> q0
a : 2, q1 -> q1
b : 0, q1 -> qd
a : 0, qd -> qd
b : 0, qd -> qd
