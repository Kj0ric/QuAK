# Threshold-extremal Max_f regression:
# - the child reaches a wrong final with return 0 on `a`
# - it can continue through that final and reach return 1 on `b`
# - threshold 1 must therefore keep both guesses available on `a`

@PARENT
final: q0
a : 1, q0 -> q1
b : 0, q1 -> q0

@CHILD 0

@CHILD 1
final: f0 f1
a : 0, s0 -> f0
b : 1, f0 -> f1
